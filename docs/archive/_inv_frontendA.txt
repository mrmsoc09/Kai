from fastapi import APIRouter, HTTPException
from typing import Dict, Any, List
from apps.backend.src.core.run_store import load_all_runs

router = APIRouter(prefix="/runs", tags=["runs"])

@router.get("")
@router.get("/")
def list_runs(limit: int = 50) -> List[Dict[str, Any]]:
    runs = load_all_runs()
    runs_sorted = sorted(runs, key=lambda r: r.get('timestamp', ''), reverse=True)
    return runs_sorted[: max(1, min(limit, 200))]

@router.get("/{run_id}")
def get_run(run_id: str) -> Dict[str, Any]:
    for r in load_all_runs():
        if r.get('run_id') == run_id:
            return r
    raise HTTPException(status_code=404, detail="run_not_found")

import React from "react";

const COLORS: Record<string, string> = {
  VALIDATED: "#ff4d4f", // Red
  LIKELY: "#fa8c16",    // Orange
  HYPOTHESIS: "#fadb14", // Yellow
  SIGNAL: "#fadb14",     // Yellow
  SAFE: "#52c41a",       // Green
  MITIGATED: "#52c41a",  // Green
  INFO: "#1677ff",       // Blue
  PARKED: "#8c8c8c",     // Gray
  BLOCKED: "#8c8c8c"     // Gray
};

type Props = { status?: string };

export default function StatusBadge({ status }: Props){
  const s = (status || "INFO").toUpperCase();
  const bg = COLORS[s] || COLORS.INFO;
  const style: React.CSSProperties = {
    display: "inline-block",
    padding: "4px 8px",
    borderRadius: 6,
    background: bg,
    color: "#000",
    fontWeight: 600,
    fontSize: 12
  };
  return (<span style={style}>{s}</span>);
}

===== apps/frontend/src/pages/Runs.tsx =====

import React, { useEffect, useState } from "react";
import RunHUD from "../components/RunHUD";
import StatusBadge from "../components/StatusBadge";

type RunRec = any;

export default function Runs(){
  const [runs, setRuns] = useState<RunRec[]>([]);
  const [sel, setSel] = useState<RunRec | null>(null);

  useEffect(()=>{
    fetch("/runs").then(r=>r.json()).then(setRuns).catch(()=>setRuns([]));
  },[]);

  return (
    <div style={{padding:16}}>
      <h2>Runs</h2>
      <div style={{display:"grid", gridTemplateColumns:"1fr 2fr", gap:16}}>
        <div>
          {(runs||[]).map((r: RunRec)=>{
            return (
              <div key={r.run_id} onClick={()=>setSel(r)} style={{border:"1px solid #eee", borderRadius:8, padding:8, marginBottom:8, cursor:"pointer"}}>
                <div style={{display:"flex", justifyContent:"space-between"}}>
                  <div><strong>{r.run_id}</strong></div>
                  <StatusBadge status={(r.findings?.[0]?.status)||"INFO"} />
                </div>
                <div style={{fontSize:12, color:"#666"}}>{r.module} • {r.mode} • {r.target}</div>
              </div>
            );
          })}
        </div>
        <div>
          <RunHUD run={sel || runs[0]} />
        </div>
      </div>
    </div>
  );
}

import React, { useEffect, useState } from "react";

type Note = { id: string; title?: string; path?: string; classification?: string; type?: string };

export default function Knowledge(){
  const [status, setStatus] = useState<any>({});
  const [notes, setNotes] = useState<Note[]>([]);
  const [count, setCount] = useState<number>(0);

  const load = ()=>{
    fetch("/knowledge/status").then(r=>r.json()).then(setStatus).catch(()=>setStatus({}));
    fetch("/knowledge/notes").then(r=>r.json()).then((d)=>{ setNotes(d.notes||[]); setCount(d.count||0); }).catch(()=>{ setNotes([]); setCount(0); });
  };

  useEffect(()=>{ load(); },[]);

  const refresh = ()=>{
    fetch("/knowledge/refresh",{method:"POST"}).then(()=>load());
  };

  return (
    <div style={{padding:16}}>
      <h2>Knowledge</h2>
      <div style={{marginBottom:12}}>
        <div>Enabled: <strong>{String(status.enabled)}</strong></div>
        <div>Vault: <code>{status.vault_path||"-"}</code></div>
        <button onClick={refresh} style={{marginTop:8}}>Refresh Vault Index</button>
      </div>
      <div>
        <h3>Notes ({count})</h3>
        {(notes||[]).map(n=> (
          <div key={n.id} style={{border:"1px solid #eee", borderRadius:8, padding:8, marginBottom:8}}>
            <div><strong>{n.title||n.id}</strong> <span style={{fontSize:12, color:"#666"}}>({n.type})</span></div>
            <div style={{fontSize:12}}>{n.classification} • <code>{n.path}</code></div>
          </div>
        ))}
      </div>
    </div>
  );
}
