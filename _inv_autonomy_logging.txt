===== core/trace.py =====

from __future__ import annotations
import os, json, datetime, re
from pathlib import Path
from typing import Dict, Any, List
import yaml

ROOT = Path(__file__).resolve().parents[4]
LOG_BASE = ROOT / 'artifacts' / 'logs'
CONF_POL = ROOT / 'configs' / 'policies.yaml'
CONF_KNOW = ROOT / 'configs' / 'knowledge.yaml'
LOG_BASE.mkdir(parents=True, exist_ok=True)

SAFE_DEFAULT_PATTERNS = [
    r"AWS_SECRET_ACCESS_KEY", r"BEGIN RSA PRIVATE KEY", r"apikey=", r"password=",
]


def _load_yaml(p: Path) -> dict:
    try:
        return yaml.safe_load(p.read_text()) or {}
    except Exception:
        return {}


def _redactors() -> List[str]:
    k = _load_yaml(CONF_KNOW)
    pats = ((k.get('governance') or {}).get('redact_patterns')) or []
    # de-duplicate
    s = set(pats + SAFE_DEFAULT_PATTERNS)
    return list(s)


def _redact_text(text: str) -> str:
    red = text
    for p in _redactors():
        try:
            red = re.sub(re.escape(p), '[REDACTED]', red)
        except re.error:
            red = red.replace(p, '[REDACTED]')
    return red


def _date_dirs() -> Path:
    now = datetime.datetime.now(datetime.timezone.utc)
    d = LOG_BASE / f"{now.year:04d}" / f"{now.month:02d}" / f"{now.day:02d}"
    d.mkdir(parents=True, exist_ok=True)
    return d


def log_paths(run_id: str) -> Dict[str, str]:
    base = _date_dirs() / 'runs' / run_id
    base.mkdir(parents=True, exist_ok=True)
    return {
        'base': str(base),
        'decision_trace': str(base / 'decision_trace.jsonl'),
        'reasoning_summary': str(base / 'reasoning_summary.md'),
        'taxonomy_tags': str(base / 'taxonomy_tags.json'),
    }


def append_decision_trace(run_id: str, entry: Dict[str, Any]) -> str:
    paths = log_paths(run_id)
    line = {
        'ts': datetime.datetime.now(datetime.timezone.utc).isoformat(),
        **{k: v for k, v in entry.items() if v is not None}
    }
    # redact string fields shallowly
    for k, v in list(line.items()):
        if isinstance(v, str):
            line[k] = _redact_text(v)
        elif isinstance(v, list):
            line[k] = [ _redact_text(i) if isinstance(i, str) else i for i in v ]
        elif isinstance(v, dict):
            # shallow redact values
            for kk, vv in list(v.items()):
                if isinstance(vv, str):
                    v[kk] = _redact_text(vv)
    with open(paths['decision_trace'], 'a', encoding='utf-8') as f:
        f.write(json.dumps(line, ensure_ascii=False) + "
")
    return paths['decision_trace']


def write_reasoning_summary(run_id: str, what_changed: str, why: str, evidence: List[str], next_actions: List[str]) -> str:
    paths = log_paths(run_id)
    md = [
        f"# Reasoning Summary — {run_id}",
        "",
        "## What changed", _redact_text(what_changed or "-"),
        "",
        "## Why", _redact_text(why or "-"),
        "",
        "## Evidence", *(f"- { _redact_text(e)}" for e in (evidence or [])),
        "",
        "## Next actions", *(f"- { _redact_text(n)}" for n in (next_actions or [])),
        "",
    ]
    with open(paths['reasoning_summary'], 'w', encoding='utf-8') as f:
        f.write("
".join(md))
    # seed empty taxonomy tags file if missing
    try:
        if not os.path.exists(paths['taxonomy_tags']):
            with open(paths['taxonomy_tags'], 'w', encoding='utf-8') as tf:
                tf.write(json.dumps({"tags": []}, ensure_ascii=False))
    except Exception:
        pass
    return paths['reasoning_summary']


def policy_gate_info(mode: str, external: bool) -> Dict[str, any]:
    pol = _load_yaml(CONF_POL) or {}
    auto = (pol.get('autonomy') or {})
    if mode != 'execute':
        tier = 0
        name = 'AUTO'
    else:
        tier = 2 if external else 0
        name = 'APPROVE' if external else 'AUTO'
    return {
        'tier': tier,
        'name': name,
        'rules': auto.get('gating_rules', [])
    }

===== routers/logs.py =====

from fastapi import APIRouter, HTTPException
from typing import List, Dict, Any
from apps.backend.src.core.trace import log_paths
import os, json

router = APIRouter(prefix='/logs', tags=['logs'])

@router.get('/{run_id}/decision_trace')
def decision_trace(run_id: str) -> List[Dict[str, Any]]:
    paths = log_paths(run_id)
    p = paths['decision_trace']
    if not os.path.exists(p):
        return []
    out = []
    with open(p, 'r', encoding='utf-8') as f:
        for line in f:
            try:
                out.append(json.loads(line))
            except Exception:
                pass
    return out

@router.get('/{run_id}/summary')
def summary(run_id: str) -> Dict[str, Any]:
    paths = log_paths(run_id)
    md = ''
    try:
        with open(paths['reasoning_summary'], 'r', encoding='utf-8') as f:
            md = f.read()
    except Exception:
        pass
    return { 'markdown': md }
