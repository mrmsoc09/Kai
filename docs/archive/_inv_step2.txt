===== core/run_store.py =====
import json
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
ART_DIR = ROOT / "artifacts" / "dork_runs"


def ensure_dir(p: Path):
    p.mkdir(parents=True, exist_ok=True)


def run_dir(run_id: str) -> Path:
    d = ART_DIR / run_id
    ensure_dir(d)
    return d


def write_run_record(run_id: str, record: dict) -> str:
    d = run_dir(run_id)
    f = d / "run.json"
    f.write_text(json.dumps(record, ensure_ascii=False, indent=2))
    return str(f)

from fastapi import APIRouter, HTTPException
from fastapi.responses import JSONResponse
from typing import Dict, Any, List
import os, yaml
from datetime import datetime

# Paths
BASE = os.path.abspath(os.path.join(os.path.dirname(__file__), "../../../.."))
LIB_DIR = os.path.join(BASE, "data", "dorks", "google")
POLICY_PATH = os.path.join(BASE, "configs", "policies.yaml")

# Helpers
try:
    from modules.scanners.google_dorks.cse_client import GoogleCSE
except Exception:
    GoogleCSE = None  # type: ignore

from apps.backend.src.core.run_store import write_run_record

router = APIRouter(prefix="/dorks", tags=["dorks"])


def _load_yaml(p: str) -> Dict[str, Any]:
    with open(p, "r") as f:
        return yaml.safe_load(f) or {}


def _policies() -> Dict[str, Any]:
    try:
        return _load_yaml(POLICY_PATH)
    except Exception:
        return {}


def _resolve_queries(target: str, query: str = None, chain: str = None) -> List[str]:
    base = _load_yaml(os.path.join(LIB_DIR, "base.yaml"))
    if chain:
        chains = _load_yaml(os.path.join(LIB_DIR, "chains.yaml"))
        cdef = next((c for c in chains.get("chains", []) if c.get("name") == chain), None)
        if not cdef:
            raise HTTPException(404, f"chain {chain} not found")
        return [step["query"].replace("{target}", target) for step in cdef.get("steps", [])]
    if query:
        return [query.replace("{target}", target)]
    cats = base.get("categories", {})
    for v in cats.values():
        return [q.replace("{target}", target) for q in v]
    return []


def _now():
    return datetime.utcnow().strftime(%Y-%m-%dT%H:%M:%SZ)


def _base_run_record(run_id: str, module: str, mode: str, target: str, planned: List[str]) -> Dict[str, Any]:
    loops = {"strategic": {"name": "GPEE", "step": "Plan"}, "tactical": {"name": "OODA", "step": "Orient/Decide"}}
    if mode == "execute":
        loops = {"strategic": {"name": "GPEE", "step": "Execute/Evaluate"}, "tactical": {"name": "OODA", "step": "Act/Observe"}}
    candidates = [
        {"name": "plan_dorks", "score_breakdown": {"impact": 3, "likelihood": 5, "chain": 3, "cost_time": 1, "scope_risk": 1}, "score_total": 9},
        {"name": "execute_cse", "score_breakdown": {"impact": 5, "likelihood": 5, "chain": 4, "cost_time": 2, "scope_risk": 1}, "score_total": 11},
        {"name": "defer", "score_breakdown": {"impact": 1, "likelihood": 3, "chain": 1, "cost_time": 1, "scope_risk": 1}, "score_total": 3}
    ]
    chosen = candidates[0] if mode != "execute" else max(candidates, key=lambda x: x["score_total"])
    record = {
        "run_id": run_id,
        "module": module,
        "target": target,
        "mode": mode,
        "timestamp": _now(),
        "loops": loops,
        "ledger": {
            "assets": [],
            "hypotheses": [{"text": "CSE-executed dorks produce high-signal OSINT safely", "score": 11}],
            "evidence": [],
            "attack_graph": {"nodes": ["DorkLibrary", "CSEClient", "AuditBundle"], "edges": [["DorkLibrary","CSEClient"],["CSEClient","AuditBundle"]]},
            "branches": {"active": ["dorks"], "parked": ["osint.connectors","nuclei"]},
            "open_questions": ["Target scope confirmation", "Frontend HUD wiring"]
        },
        "decision": {"candidates": candidates, "chosen": {"name": chosen["name"], "score_total": chosen["score_total"], "why": "highest score"}},
        "approval": {"hil_required": mode == "execute", "hil_approved": False, "approver": None, "approved_at": None},
        "artifacts": {"audit_bundle_path": None, "recording_path": None, "screenshots": []},
        "planned_queries": planned,
        "results": []
    }
    return record


@router.get("/categories")
def categories() -> Dict[str, Any]:
    data = _load_yaml(os.path.join(LIB_DIR, "base.yaml"))
    return {"parameters": data.get("parameters", {}), "categories": data.get("categories", {})}


@router.get("/chains")
def chains() -> Dict[str, Any]:
    return _load_yaml(os.path.join(LIB_DIR, "chains.yaml"))


@router.post("/run")
def run(payload: Dict[str, Any]) -> Dict[str, Any]:
    target = payload.get("target")
    if not target:
        raise HTTPException(400, "target required")
    mode = (payload.get("mode") or "plan").lower()
    query = payload.get("query")
    chain = payload.get("chain")
    limit = int(payload.get("limit") or 5)

    planned = _resolve_queries(target, query, chain)

    rid = f"dorks-{datetime.utcnow().strftime(%Y%m%d%H%M%S)}"
    record = _base_run_record(rid, module="scanners.google_dorks", mode=mode, target=target, planned=planned)

    if mode != "execute":
        run_path = write_run_record(rid, record)
        return {"mode": "plan", "run_id": rid, "executed": planned, "run_record": run_path, "note": "No external queries performed."}

    policies = _policies()
    hil_approved = bool(payload.get("hil_approved") is True)
    external_enabled = bool(policies.get("external_queries_enabled") is True)

    if not hil_approved:
        return JSONResponse(status_code=409, content={
            "status": "blocked", "reason": "hil_approval_required",
            "next": "use mode=plan or set hil_approved=true after human review"
        })
    if not external_enabled:
        return JSONResponse(status_code=409, content={
            "status": "blocked", "reason": "policy_external_block",
            "next": "enable external_queries_enabled=true in configs/policies.yaml"
        })
    if GoogleCSE is None:
        raise HTTPException(500, "Executor unavailable")

    cse = GoogleCSE()
    results: List[Dict[str, Any]] = []
    for q in planned:
        out = cse.search(q, num=limit)
        results.append({"query": q, "result": out})

    from_path = record["artifacts"].get("audit_bundle_path")
    audit_path = cse.audit_log(rid, planned, results, {"target": target, "policy": policies})
    record["results"] = results
    record["artifacts"]["audit_bundle_path"] = audit_path
    record["approval"]["hil_approved"] = True
    record["approval"]["approver"] = "human"
    record["approval"]["approved_at"] = _now()

    run_path = write_run_record(rid, record)
    return {"mode": "execute", "run_id": rid, "planned": planned, "results": results, "audit": audit_path, "run_record": run_path}
