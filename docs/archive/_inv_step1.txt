from fastapi import APIRouter, HTTPException
from fastapi.responses import JSONResponse
from typing import Dict, Any, List
import os, yaml
from datetime import datetime

# Library paths
BASE = os.path.abspath(os.path.join(os.path.dirname(__file__), "../../../.."))
LIB_DIR = os.path.join(BASE, "data", "dorks", "google")
POLICY_PATH = os.path.join(BASE, "configs", "policies.yaml")

# Optional executor (only used when gating conditions pass)
try:
    from modules.scanners.google_dorks.cse_client import GoogleCSE
except Exception:
    GoogleCSE = None  # type: ignore

router = APIRouter(prefix="/dorks", tags=["dorks"])


def _load_yaml(p: str) -> Dict[str, Any]:
    with open(p, "r") as f:
        return yaml.safe_load(f) or {}


def _policies() -> Dict[str, Any]:
    try:
        return _load_yaml(POLICY_PATH)
    except Exception:
        return {}


def _resolve_queries(target: str, query: str = None, chain: str = None) -> List[str]:
    base = _load_yaml(os.path.join(LIB_DIR, "base.yaml"))
    if chain:
        chains = _load_yaml(os.path.join(LIB_DIR, "chains.yaml"))
        cdef = next((c for c in chains.get("chains", []) if c.get("name") == chain), None)
        if not cdef:
            raise HTTPException(404, f"chain {chain} not found")
        return [step["query"].replace("{target}", target) for step in cdef.get("steps", [])]
    if query:
        return [query.replace("{target}", target)]
    cats = base.get("categories", {})
    for v in cats.values():
        return [q.replace("{target}", target) for q in v]
    return []


@router.get("/categories")
def categories() -> Dict[str, Any]:
    data = _load_yaml(os.path.join(LIB_DIR, "base.yaml"))
    return {"parameters": data.get("parameters", {}), "categories": data.get("categories", {})}


@router.get("/chains")
def chains() -> Dict[str, Any]:
    return _load_yaml(os.path.join(LIB_DIR, "chains.yaml"))


@router.post("/run")
def run(payload: Dict[str, Any]) -> Dict[str, Any]:
    target = payload.get("target")
    if not target:
        raise HTTPException(400, "target required")
    mode = (payload.get("mode") or "plan").lower()
    query = payload.get("query")
    chain = payload.get("chain")
    limit = int(payload.get("limit") or 5)

    planned = _resolve_queries(target, query, chain)

    # Plan mode: always safe, no external calls
    if mode != "execute":
        return {
            "mode": "plan",
            "executed": planned,
            "note": "No external queries performed. Use mode=execute with required approvals to run via Google CSE.",
        }

    # Execute mode: enforce HiL + policy gates
    policies = _policies()
    hil_approved = bool(payload.get("hil_approved") is True)
    external_enabled = bool(policies.get("external_queries_enabled") is True)

    if not hil_approved:
        return JSONResponse(status_code=409, content={
            "status": "blocked",
            "reason": "hil_approval_required",
            "next": "use mode=plan or set hil_approved=true after human review"
        })

    if not external_enabled:
        return JSONResponse(status_code=409, content={
            "status": "blocked",
            "reason": "policy_external_block",
            "next": "enable external_queries_enabled=true in configs/policies.yaml"
        })

    if GoogleCSE is None:
        raise HTTPException(500, "Executor unavailable")

    cse = GoogleCSE()
    results: List[Dict[str, Any]] = []
    for q in planned:
        out = cse.search(q, num=limit)
        results.append({"query": q, "result": out})
    run_id = f"dorks-{datetime.utcnow().strftime(%Y%m%d%H%M%S)}"
    audit_path = cse.audit_log(run_id, planned, results, {"target": target, "policy": policies})
    return {"mode": "execute", "planned": planned, "results": results, "audit": audit_path}

from fastapi.testclient import TestClient
from apps.backend.src.main import app

client = TestClient(app)

def test_plan_mode_is_safe():
    r = client.post("/dorks/run", json={"target":"example.com","mode":"plan","chain":"backup_index_chain"})
    assert r.status_code == 200
    body = r.json()
    assert body["mode"] == "plan"
    assert "executed" in body

def test_execute_requires_hil_approval():
    r = client.post("/dorks/run", json={"target":"example.com","mode":"execute"})
    assert r.status_code == 409
    body = r.json()
    assert body.get("reason") == "hil_approval_required"

def test_execute_policy_blocked_by_default():
    r = client.post("/dorks/run", json={"target":"example.com","mode":"execute","hil_approved": True})
    assert r.status_code == 409
    body = r.json()
    assert body.get("reason") == "policy_external_block"
